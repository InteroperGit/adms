# PDR: Sitemap generation — SSG script + SSR endpoint

**Date**: 2026-03-24

## Goal

Generate a valid `sitemap.xml` in two modes:

1. **SSG (Static)** — a post-build script `scripts/postbuild-sitemap.ts` that
   reads the same route list produced by `react-router.config.ts → prerender()`
   and writes `build/client/sitemap.xml` after the build.
2. **SSR (Dynamic)** — a React Router resource route `src/routes/sitemap.xml.ts`
   that returns a fresh `sitemap.xml` response at runtime, including any routes
   that are only known at request time (e.g. portfolio cases fetched from API).

Both modes share a single `src/lib/sitemap.ts` helper that builds the XML string
from a list of URL entries.

---

## Route inventory (current)

Derived from `react-router.config.ts → prerender()`:

| Path pattern | Source |
|---|---|
| `/` | `getStaticPaths()` |
| `/portfolio` | static |
| `/portfolio/:catSlug` | `categories.json` |
| `/portfolio/:catSlug/:year/:month/:slug` | `data/content/portfolio/**/*.json` |
| `/404` | static |
| Legal pages (e.g. `/privacy`, `/terms`) | `getStaticPaths()` |

The SSR route must reconstruct this list from the same data sources at request time.

## Changes

### 1. `src/lib/sitemap.ts` (new)

Shared helper — no framework dependency, pure string builder.

```ts
export interface SitemapEntry {
  loc: string;       // absolute URL
  lastmod?: string;  // YYYY-MM-DD
  changefreq?: 'always' | 'hourly' | 'daily' | 'weekly' | 'monthly' | 'yearly' | 'never';
  priority?: number; // 0.0–1.0
}

export function buildSitemapXml(entries: SitemapEntry[]): string {
  const urls = entries
    .map((e) => {
      const loc = `  <loc>${e.loc}</loc>`;
      const lastmod = e.lastmod ? `\n  <lastmod>${e.lastmod}</lastmod>` : '';
      const changefreq = e.changefreq ? `\n  <changefreq>${e.changefreq}</changefreq>` : '';
      const priority = e.priority != null ? `\n  <priority>${e.priority.toFixed(1)}</priority>` : '';
      return `<url>\n${loc}${lastmod}${changefreq}${priority}\n</url>`;
    })
    .join('\n');
  return `<?xml version="1.0" encoding="UTF-8"?>\n<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">\n${urls}\n</urlset>`;
}
```

### 2. `scripts/postbuild-sitemap.ts` (new)

Post-build script for SSG mode. Runs after `postbuild-seo.ts` in the build chain.

```ts
/**
 * Post-build script: generate sitemap.xml for SSG builds.
 * Reads the same data sources as react-router.config.ts prerender() and
 * writes build/client/sitemap.xml.
 *
 * Usage:
 *   react-router build && vite-node scripts/postbuild-sitemap.ts
 *
 * Env vars:
 *   SITE_URL  Base URL (default: https://example.com)
 */

import { fileURLToPath } from 'url';
import { dirname, resolve } from 'path';
import { writeFileSync, readFileSync } from 'fs';
import { readJson, walkJsonFiles, extractYearMonth } from './buildUtils';
import { buildSitemapXml, type SitemapEntry } from '../src/lib/sitemap';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);
const rootDir = resolve(__dirname, '..');
const distDir = resolve(rootDir, 'build/client');

const BASE_URL = (process.env.SITE_URL ?? 'https://example.com').replace(/\/$/, '');

const colors = {
  reset: '\x1b[0m',
  green: '\x1b[32m',
  cyan: '\x1b[36m',
  yellow: '\x1b[33m',
};

function log(msg: string) {
  console.log(`${colors.cyan}[sitemap]${colors.reset} ${msg}`);
}

interface CategoryEntry { name: string; slug: string; }
interface CaseEntry { slug: string; category: string; publishDate: string; }

const today = new Date().toISOString().slice(0, 10);

function buildEntries(): SitemapEntry[] {
  const categories =
    readJson<CategoryEntry[]>(resolve(rootDir, 'data/content/config/categories.json')) ?? [];
  const catSlugs = ['all', ...categories.map((c) => c.slug)];

  const portfolioDir = resolve(rootDir, 'data/content/portfolio');
  const cases = walkJsonFiles(portfolioDir).map(
    (f) => JSON.parse(readFileSync(f, 'utf-8')) as CaseEntry
  );

  const staticRoutes: SitemapEntry[] = [
    { loc: `${BASE_URL}/`,         changefreq: 'weekly',  priority: 1.0, lastmod: today },
    { loc: `${BASE_URL}/portfolio`, changefreq: 'daily',   priority: 0.9, lastmod: today },
  ];

  const categoryRoutes: SitemapEntry[] = catSlugs.map((s) => ({
    loc: `${BASE_URL}/portfolio/${s}`,
    changefreq: 'daily',
    priority: 0.8,
    lastmod: today,
  }));

  const caseRoutes: SitemapEntry[] = cases.flatMap((c) => {
    const { year, month } = extractYearMonth(c.publishDate);
    const catSlug = categories.find((cat) => cat.name === c.category)?.slug;
    const entry = (path: string): SitemapEntry => ({
      loc: `${BASE_URL}${path}`,
      changefreq: 'monthly',
      priority: 0.7,
      lastmod: c.publishDate.slice(0, 10),
    });
    const allPath = `/portfolio/all/${year}/${month}/${c.slug}`;
    return catSlug
      ? [entry(allPath), entry(`/portfolio/${catSlug}/${year}/${month}/${c.slug}`)]
      : [entry(allPath)];
  });

  return [...staticRoutes, ...categoryRoutes, ...caseRoutes];
}

try {
  const start = Date.now();
  log('Generating sitemap.xml...');
  const entries = buildEntries();
  const xml = buildSitemapXml(entries);
  writeFileSync(resolve(distDir, 'sitemap.xml'), xml, 'utf-8');
  log(`${colors.green}✓${colors.reset} ${entries.length} URLs written to build/client/sitemap.xml in ${Date.now() - start}ms`);
} catch (err) {
  console.error(`${colors.yellow}[sitemap] Error:${colors.reset}`, err);
  process.exit(1);
}
```

### 3. `src/routes/sitemap[.xml].ts` (new) — SSR resource route

Resource route that returns `sitemap.xml` at runtime. Only active when `ssr: true`
in `react-router.config.ts`. For SSG builds this file is ignored (prerendering
does not call resource routes), so it is safe to ship regardless.

```ts
import type { Route } from './+types/sitemap[.xml]';
import { readFileSync } from 'fs';
import path from 'path';
import { buildSitemapXml, type SitemapEntry } from '~/lib/sitemap';

const BASE_URL = (process.env.SITE_URL ?? 'https://example.com').replace(/\/$/, '');

export async function loader({ context }: Route.LoaderArgs) {
  // Re-use the same data-reading logic as the SSG script.
  // In SSR the data directory is co-located with the server bundle.
  const root = process.cwd();

  // Read categories
  const categoriesPath = path.resolve(root, 'data/content/config/categories.json');
  const categories: { name: string; slug: string }[] = JSON.parse(
    readFileSync(categoriesPath, 'utf-8')
  );
  const catSlugs = ['all', ...categories.map((c) => c.slug)];

  // Read portfolio cases
  const portfolioDir = path.resolve(root, 'data/content/portfolio');
  // walkJsonFiles is a build-time util — in SSR use fs.readdirSync recursively
  const { walkJsonFiles, extractYearMonth } = await import('../../scripts/buildUtils');
  const cases: { slug: string; category: string; publishDate: string }[] =
    walkJsonFiles(portfolioDir).map((f) => JSON.parse(readFileSync(f, 'utf-8')));

  const today = new Date().toISOString().slice(0, 10);

  const entries: SitemapEntry[] = [
    { loc: `${BASE_URL}/`,          changefreq: 'weekly',  priority: 1.0, lastmod: today },
    { loc: `${BASE_URL}/portfolio`, changefreq: 'daily',   priority: 0.9, lastmod: today },
    ...catSlugs.map((s) => ({
      loc: `${BASE_URL}/portfolio/${s}`,
      changefreq: 'daily' as const,
      priority: 0.8,
      lastmod: today,
    })),
    ...cases.flatMap((c) => {
      const { year, month } = extractYearMonth(c.publishDate);
      const catSlug = categories.find((cat) => cat.name === c.category)?.slug;
      const entry = (p: string): SitemapEntry => ({
        loc: `${BASE_URL}${p}`,
        changefreq: 'monthly',
        priority: 0.7,
        lastmod: c.publishDate.slice(0, 10),
      });
      const allPath = `/portfolio/all/${year}/${month}/${c.slug}`;
      return catSlug
        ? [entry(allPath), entry(`/portfolio/${catSlug}/${year}/${month}/${c.slug}`)]
        : [entry(allPath)];
    }),
  ];

  return new Response(buildSitemapXml(entries), {
    headers: {
      'Content-Type': 'application/xml; charset=utf-8',
      'Cache-Control': 'public, max-age=3600',
    },
  });
}
```

### 4. `package.json` — extend `build` script

Append the sitemap script to the existing build pipeline:

```diff
- "build": "tsc -b tsconfig.app.json && react-router build && vite-node scripts/postbuild-seo.ts && vite-node scripts/postbuild-cache.ts",
+ "build": "tsc -b tsconfig.app.json && react-router build && vite-node scripts/postbuild-seo.ts && vite-node scripts/postbuild-cache.ts && vite-node scripts/postbuild-sitemap.ts",
```

### 5. `react-router.config.ts` — add `sitemap.xml` to prerendered routes (SSG)

For SSG, also prerender the sitemap URL so it is reachable as a static file.
This is only relevant if `ssr: false` and the sitemap is served from a CDN.
Skip if `sitemap.xml` is already produced by the post-build script directly.

> **Decision**: Since `postbuild-sitemap.ts` writes `build/client/sitemap.xml`
> directly, no change to `react-router.config.ts` is needed for SSG mode.
> The file is available on the CDN without going through a route.
